  if(!navigator.gpu){say("NO-GPU");}else{
    const a=await navigator.gpu.requestAdapter({powerPreference:"high-performance"});
    say("adapter="+!!a);
    if(a){
      say("info="+JSON.stringify(a.info??{}));
      say("features="+[...a.features].join(","));
      const d=await a.requestDevice({requiredFeatures:[...a.features].filter(f=>f==="timestamp-query"||f==="shader-f16")});
      say("device="+!!d);
      say("queue="+!!d.queue);
      say("DONE-OK");
    } else say("NO-ADAPTER");
  }
}catch(e){say("THREW "+e);}
