

# ===========================================================================
# THE GATE.
#
# EVERY `py=` LITERAL BELOW IS A ROW OF `.agents/slop/ga-oracle.txt`, WHICH
# CALLS generate.py's OWN FUNCTIONS -- `_strip_enc`, `_norm_field`, `_map_flat`,
# and the three emitters, whose emitted FILES are read back off disk.  No `py=`
# here was typed by a human; `.agents/slop/ga_fix.py` splices them in from the
# oracle and REFUSES to emit a row whose name is not in the oracle file.  That
# is the rule five files in this project broke by hand (`cstyle` 17 of 215,
# `ops_nv` 33 of 219, `rdna` one hex digit, `amdev` 7 of 80, `ops_qcom` one
# table entry).
#
# THE EMITTER ROWS ARE ONE ROW PER EMITTED FILE, THE FILE JOINED BY NEWLINE.
# A count row would be identical for a dropped member, a dropped alias, a
# swapped `default=NULL` and a reordered field; a string diff is not.  This is
# renderer convention 2 applied to a GENERATOR: the answer IS the file.
# ===========================================================================

def g(nm: String, got: String, want: String) -> IO(Unit):
  IO.print(String.concat([nm, " = [", got, "]   py=[", want, "]"]))

def gl(nm: String, got: List<&2, String>, want: String) -> IO(Unit):
  g(nm, String.join(got, "\n"), want)

def main() -> IO(Unit):
  do IO<Unit>:
    _ : Unit <- IO.print("x")
