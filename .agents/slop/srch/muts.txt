# Mutations for tinybendygrad/codegen/opt/search.bend, one stanza each. `LINE` is an
# exact substring of the target line and must match EXACTLY ONE line, so a stanza that
# has gone stale is a loud failure rather than a silent no-op.

# Group 1 -- the eight group sizes, one mutation each. Each must move its OWN row by
# name, not only the two totals.
--- grp.up 60 -> 61
LINE def grp.up() -> U32: 60
NEW  def grp.up() -> U32: 61
--- grp.lcl 64 -> 63
LINE def grp.lcl() -> U32: 64
NEW  def grp.lcl() -> U32: 63
--- grp.lcl_top 64 -> 63
LINE def grp.lcl_top() -> U32: 64
NEW  def grp.lcl_top() -> U32: 63
--- grp.padto True 7 -> 8
LINE case True{}: 7
NEW  case True{}: 8
--- grp.split32 1 -> 2
LINE def grp.split32() -> U32: 1
NEW  def grp.split32() -> U32: 2
--- grp.tc0 1 -> 2
LINE def grp.tc0() -> U32: 1
NEW  def grp.tc0() -> U32: 2
--- grp.tc 9 -> 8
LINE def grp.tc() -> U32: 9
NEW  def grp.tc() -> U32: 8
--- grp.swap 10 -> 9
LINE def grp.swap() -> U32: 10
NEW  def grp.swap() -> U32: 9

# Group 2 -- THE TOTAL-PRESERVING RESHUFFLES. Each of these LEAVES `acts_n` at 209, and
# each must still move a row. This group is the whole reason the census exists, and each
# stanza is one line precisely so that the compensating change is not needed: a mutation
# that edits two numbers to keep a total honest is a mutation that can hide which of the
# two the total was hiding.
--- the 60 UPCAST entries re-attributed to LOCAL, total held at 209
LINE def tgt_up() -> U32: grp.up()
NEW  def tgt_up() -> U32: U32.add(grp.lcl(), grp.lcl_top())
--- the 10 SWAP entries re-attributed to TC, total held at 209
LINE def byop.tc() -> U32:
NEW  def byop.tc() -> U32: U32.add(grp.swap(), grp.tc())
--- tgt_kinds 2 -> 3 (a rebase that reinstates UNROLL as a third target)
LINE def tgt_kinds() -> U32: 2
NEW  def tgt_kinds() -> U32: 3

# Group 3 -- the amt-0 census, which `zero_variant.flat` NOW READS. Because count and
# predicate share one source, a census change moves the ladder too; that coupling is the
# fix for the drift that let a deleted UNROLL group survive in `acts_zero` alone.
--- zero_n.up 10 -> 11, count and ladder together
LINE def zero_n.up() -> U32: 10
NEW  def zero_n.up() -> U32: 11
--- zero_n.lcl 8 -> 7
LINE def zero_n.lcl() -> U32: 8
NEW  def zero_n.lcl() -> U32: 7
--- zero_top_n 0 -> 1 (a table that grew an amt-0 entry in the top=True group)
LINE def zero_top_n() -> U32: 0
NEW  def zero_top_n() -> U32: 1

# Group 4 -- THE OFF-BY-ONE I ACTUALLY MADE AND FIXED. `zero_max.*` is the highest axis
# that HAS a twin; the ladder needs the exclusive bound, which is `zero_n`. Feeding
# `zero_max` to the ladder narrows that ladder by one axis.
--- UPCAST ladder reads zero_max instead of zero_n
LINE case O.AXIS_UPCAST{}: U32.is_lt(ax, zero_n.up())
NEW  case O.AXIS_UPCAST{}: U32.is_lt(ax, zero_max.up())
--- LOCAL ladder reads zero_max instead of zero_n
LINE case O.AXIS_LOCAL{}: U32.is_lt(ax, zero_n.lcl())
NEW  case O.AXIS_LOCAL{}: U32.is_lt(ax, zero_max.lcl())

# Group 5 -- the ladder's arms.
--- top=True arm allows the flat ladder
LINE   case True{}: False{}
NEW  case True{}: zero_variant.flat(t, ax)
--- the dead arm widens: a deleted member answers as if it were LOCAL
LINE case O.AXIS_REDUCE{}: False{}
NEW  case O.AXIS_REDUCE{}: U32.is_lt(ax, zero_n.lcl())

# Group 6 -- the beam rows, untouched by this change but still live.
--- drop_ok inverted
LINE Bool.or(Bool.not(ok), seen)
NEW  Bool.or(ok, seen)
--- min2 returns the larger
LINE   case True{}: a
NEW  case True{}: b
--- plus_dropped subtracts 3 instead of 4, so `dropped` counts a survivor that is not there
LINE U32.sub(4, U32.from_nat(List.length(&2, SCORE, plus_scores())))
NEW  U32.sub(3, U32.from_nat(List.length(&2, SCORE, plus_scores())))
